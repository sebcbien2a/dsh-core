_HASH_PATTERN = re.compile(r"[0-9a-fA-F]{64}", re.ASCII)
_TRANSACTION_PHASES = {
    "PREPARED",
    "REPLACING",
    "COMMITTED",
    "ROLLED_BACK",
    "RECOVERY_REQUIRED",
}
_RESOLVED_PHASES = {"COMMITTED", "ROLLED_BACK"}
_STATE_FILES = (
    "accounting/rows.json",
    "accounting/master_version.json",
    "banking/rows.json",
    "banking/master_version.json",
    "audit/events.jsonl",
    "audit/status_history.jsonl",
)

    ("processing_status", StatusDomain.PROCESSING),

    ("current_file_state", StatusDomain.FILE_STATE),
)
_BANKING_STATUSES = (
    ("import_status", StatusDomain.INGESTION),
    ("banking_record.reconciliation_status", StatusDomain.RECONCILIATION),
)


class LocalMasterWriterError(Exception):
    """Base exception for local master writer operations."""


class LocalMasterInitializationError(LocalMasterWriterError):
    """Raised when the local master tables cannot be initialized safely."""


class LocalMasterCorruptionError(LocalMasterWriterError):
    """Raised when persisted local master data is invalid or incomplete."""


class LocalMasterAtomicWriteError(LocalMasterWriterError):
    """Raised when a local transaction cannot be completed or rolled back."""


class LocalMasterRowConflictError(LocalMasterWriterError):
    """Raised when a row update has a conflicting or invalid version."""


class LocalMasterRecoveryError(LocalMasterWriterError):
    """Raised when an interrupted write cannot be safely recovered."""


class LocalMasterWriteResult(BaseModel):
    model_config = ConfigDict(extra="forbid")

    master_type: MasterType
    operation_id: UUID
    previous_master_version: StrictInt
    new_master_version: StrictInt
    created_row_ids: list[UUID] = Field(default_factory=list)
    updated_row_ids: list[UUID] = Field(default_factory=list)
    unchanged_row_ids: list[UUID] = Field(default_factory=list)
    audit_event_ids: list[UUID] = Field(default_factory=list)
    status_history_ids: list[UUID] = Field(default_factory=list)
    committed_at: datetime
    content_hash: str
    idempotency_key: str

    @field_validator(
        "previous_master_version",
        "new_master_version",
    )
    @classmethod
    def validate_versions(cls, value: int) -> int:
        if value < 0:
            raise ValueError("master versions must be non-negative")
        return value

    @field_validator("committed_at")
    @classmethod
    def validate_committed_at(cls, value: datetime) -> datetime:
        if value.tzinfo is None or value.utcoffset() != timedelta(0):
            raise ValueError("committed_at must be timezone-aware UTC")
        return value

    @field_validator("content_hash")
    @classmethod
    def validate_content_hash(cls, value: str) -> str:
        if not _HASH_PATTERN.fullmatch(value):
            raise ValueError("content_hash must be a SHA-256 hex digest")
        return value

    @field_validator("idempotency_key")
    @classmethod
    def validate_idempotency_key(cls, value: str) -> str:
        if not value.strip():
            raise ValueError("idempotency_key must not be blank")
        return value

    @model_validator(mode="after")
    def validate_result(self) -> LocalMasterWriteResult:
        if self.new_master_version != self.previous_master_version + 1:
            raise ValueError("new master version must increment by one")
        lists = (
            self.created_row_ids,
            self.updated_row_ids,
            self.unchanged_row_ids,
            self.audit_event_ids,
            self.status_history_ids,
        )
        all_ids = [item for values in lists for item in values]
        if len(all_ids) != len(set(all_ids)):
            raise ValueError("result identifiers must be unique across lists")
        return self


class LocalTransactionManifest(BaseModel):
    model_config = ConfigDict(extra="forbid")

    transaction_manifest_id: UUID
    master_type: MasterType
    operation_id: UUID
    idempotency_key: str
    phase: str
    created_at: datetime
    updated_at: datetime
    target_files: list[str]
    backup_files: list[str]
    content_hashes: dict[str, str]
    metadata: dict[str, Any] = Field(default_factory=dict)

    @field_validator("phase")
    @classmethod
    def validate_phase(cls, value: str) -> str:
        if value not in _TRANSACTION_PHASES:
            raise ValueError("invalid transaction phase")
        return value

    @field_validator("created_at", "updated_at")
    @classmethod
    def validate_utc(cls, value: datetime) -> datetime:
        if value.tzinfo is None or value.utcoffset() != timedelta(0):
            raise ValueError("manifest timestamps must be timezone-aware UTC")
        return value

    @field_validator("target_files", "backup_files")
    @classmethod
    def validate_relative_paths(cls, paths: list[str]) -> list[str]:
        for value in paths:
            path = PurePosixPath(value)
            if (
                not value
                or path.is_absolute()
                or "\\" in value
                or any(part == ".." or ":" in part for part in path.parts)
            ):
                raise ValueError("manifest paths must be relative POSIX paths")
        return paths


class LocalMasterWriter:
    def __init__(self, root_path: str | Path):
        self._root = Path(root_path)

    @staticmethod
    def _master_name(master_type: MasterType) -> str:
        return "accounting" if master_type == MasterType.ACCOUNTING else "banking"

    def _relative(self, master_type: MasterType, filename: str) -> str:
        return f"{self._master_name(master_type)}/{filename}"

    def _path(self, relative: str) -> Path:
        pure = PurePosixPath(relative)
        if (
            not relative
            or pure.is_absolute()
            or "\\" in relative
            or any(part == ".." or ":" in part for part in pure.parts)
        ):
            raise LocalMasterWriterError("path must be relative to the local master root")
        candidate = self._root.joinpath(*pure.parts)
        if not candidate.resolve().is_relative_to(self._root.resolve()):
            raise LocalMasterWriterError("path resolves outside the local master root")
        return candidate

    @staticmethod
    def _canonical_bytes(value: Any) -> bytes:
        return json.dumps(
            value,
            sort_keys=True,
            ensure_ascii=False,
            separators=(",", ":"),
        ).encode("utf-8")

    @classmethod
    def _hash_value(cls, value: Any) -> str:
        return hashlib.sha256(cls._canonical_bytes(value)).hexdigest()

    @staticmethod
    def _model_json(model: BaseModel) -> dict[str, Any]:
        return model.model_dump(mode="json")

    @staticmethod
    def _ensure_no_absolute_paths(value: Any) -> None:
        if isinstance(value, dict):
            for item in value.values():
                LocalMasterWriter._ensure_no_absolute_paths(item)
        elif isinstance(value, list):
            for item in value:
                LocalMasterWriter._ensure_no_absolute_paths(item)
        elif isinstance(value, str):
            LocalMasterWriter._reject_unsafe_reference(value)

    @staticmethod
    def _reject_unsafe_reference(value: str) -> None:
        remainder = value.split("://", 1)[1] if "://" in value else value
        segments = re.split(r"[\\/]", remainder)
        if (
            re.match(r"^[A-Za-z]:[\\/]", value)
            or value.startswith("\\\\")
            or value.lower().startswith("file:")
            or (value.startswith("/") and "://" not in value)
            or (len(segments) > 1 and ".." in segments)
        ):
            raise LocalMasterWriterError(
                "absolute local paths and parent segments are not permitted in master data"
            )

    @staticmethod
    def _exception_message(operation: str) -> str:
        return f"local master {operation} failed"

    def _read_bytes(self, relative: str) -> bytes:
        try:
            return self._path(relative).read_bytes()
        except OSError as exc:
            raise LocalMasterCorruptionError(
                self._exception_message("file read")
            ) from exc

    def _read_json(self, relative: str) -> Any:
        try:
            return json.loads(self._read_bytes(relative))
        except LocalMasterCorruptionError:
            raise
        except (UnicodeDecodeError, json.JSONDecodeError) as exc:
            raise LocalMasterCorruptionError(
                self._exception_message("JSON deserialization")
            ) from exc

    def _read_model(self, relative: str, model_type: type[BaseModel]) -> Any:
        try:
            return model_type.model_validate(self._read_json(relative))
        except LocalMasterCorruptionError:
            raise
        except Exception as exc:
            raise LocalMasterCorruptionError(
                self._exception_message("model validation")
            ) from exc

    def _read_rows(
        self,
        master_type: MasterType,
        include_deleted: bool,
    ) -> list[AccountingDocumentMasterRow] | list[BankTransactionMasterRow]:
        relative = self._relative(master_type, "rows.json")
        raw = self._read_json(relative)
        if not isinstance(raw, list):
            raise LocalMasterCorruptionError("master rows must be a JSON array")
        model_type: type[BaseModel] = (
            AccountingDocumentMasterRow
            if master_type == MasterType.ACCOUNTING
            else BankTransactionMasterRow
        )
        try:
            rows = [model_type.model_validate(item) for item in raw]
        except Exception as exc:
            raise LocalMasterCorruptionError(
                self._exception_message("row validation")
            ) from exc
        if not include_deleted:
            rows = [row for row in rows if not row.is_deleted]
        return rows  # type: ignore[return-value]

    def _read_jsonl(self, relative: str, model_type: type[BaseModel]) -> list[Any]:
        raw = self._read_bytes(relative)
        result = []
        try:
            for line in raw.splitlines():
                if not line.strip():
                    continue
                result.append(model_type.model_validate_json(line))
        except Exception as exc:
            raise LocalMasterCorruptionError(
                self._exception_message("JSONL validation")
            ) from exc
        return result

    @staticmethod
    def _write_fsynced(path: Path, data: bytes) -> None:
        path.parent.mkdir(parents=True, exist_ok=True)
        with path.open("wb") as stream:
            stream.write(data)
            stream.flush()
            os.fsync(stream.fileno())

    def _atomic_write(self, path: Path, data: bytes) -> None:
        temporary_name: str | None = None
        try:
            path.parent.mkdir(parents=True, exist_ok=True)
            with tempfile.NamedTemporaryFile(
                mode="wb",
                dir=path.parent,
                prefix=".local-master-",
                delete=False,
            ) as stream:
                temporary_name = stream.name
                stream.write(data)
                stream.flush()
                os.fsync(stream.fileno())
            os.replace(temporary_name, path)
        except OSError as exc:
            if temporary_name is not None:
                try:
                    Path(temporary_name).unlink(missing_ok=True)
                except OSError:
                    pass
            raise LocalMasterAtomicWriteError(
                self._exception_message("atomic file write")
            ) from exc

    def _write_manifest(self, manifest: LocalTransactionManifest) -> None:
        relative = f"recovery/{manifest.transaction_manifest_id}.json"
        self._atomic_write(
            self._path(relative),
            self._canonical_bytes(self._model_json(manifest)),
        )

    def _restore_manifest(self, manifest: LocalTransactionManifest) -> None:
        backup_dir = f"recovery/{manifest.transaction_manifest_id}/backup"
        missing = set(manifest.metadata.get("originally_missing", []))
        for target in manifest.target_files:
            target_path = self._path(target)
            if target in missing:
                target_path.unlink(missing_ok=True)
                continue
            backup = f"{backup_dir}/{target}"
            backup_path = self._path(backup)
            if not backup_path.is_file():
                raise LocalMasterRecoveryError(
                    "transaction backup is missing"
                )
            self._atomic_write(target_path, backup_path.read_bytes())

    @staticmethod
    def _replacement_rank(relative: str) -> tuple[int, str]:
        # Rows first, audit next, lock after, and master_version.json last so
        # that the version file acts as the final commit marker of a publication.
        name = PurePosixPath(relative).name
        rank = {
            "rows.json": 0,
            "events.jsonl": 1,
            "status_history.jsonl": 1,
            "lock.json": 2,
            "master_version.json": 3,
        }.get(name, 4)
        return rank, relative

    def _ensure_no_unresolved_transaction(self) -> None:
        recovery_dir = self._path("recovery")
        if not recovery_dir.exists():
            return
        for manifest_path in sorted(recovery_dir.glob("*.json")):
            try:
                manifest = LocalTransactionManifest.model_validate_json(
                    manifest_path.read_bytes()
                )
            except Exception as exc:
                raise LocalMasterRecoveryError("transaction manifest is corrupt") from exc
            if manifest.phase not in _RESOLVED_PHASES:
                raise LocalMasterRecoveryError(
                    "an unresolved local transaction must be recovered first"
                )

    def _verify_rows_hash(
        self,
        master_type: MasterType,
        rows: list[Any],
        version: MasterVersion,
    ) -> None:
        payloads = [
            row.model_dump(mode="json")
            for row in sorted(rows, key=lambda row: str(self._row_id(master_type, row)))
        ]
        if self._hash_value(payloads) != version.content_hash.lower():
            raise LocalMasterCorruptionError(
                "master rows do not match the master version content hash"
            )

    def _verify_integrity(self, master_type: MasterType) -> None:
        self._verify_rows_hash(
            master_type,
            self._read_rows(master_type, include_deleted=True),
            self.read_master_version(master_type),
        )

    def _transaction(
        self,
        *,
        master_type: MasterType,
        operation_id: UUID,
        idempotency_key: str,
        now: datetime,
        files: dict[str, bytes],
    ) -> UUID:
        self._ensure_no_unresolved_transaction()
        manifest_id = uuid4()
        targets = sorted(files, key=self._replacement_rank)
        recovery_base = f"recovery/{manifest_id}"
        staging_base = f"staging/{manifest_id}"
        backup_files: list[str] = []
        originally_missing: list[str] = []
        hashes = {key: hashlib.sha256(value).hexdigest() for key, value in files.items()}
        try:
            for target in targets:
                path = self._path(target)
                if path.exists():
                    backup = f"{recovery_base}/backup/{target}"
                    self._write_fsynced(self._path(backup), path.read_bytes())
                    backup_files.append(backup)
                else:
                    originally_missing.append(target)
                self._write_fsynced(
                    self._path(f"{staging_base}/new/{target}"),
                    files[target],
                )

            manifest = LocalTransactionManifest(
                transaction_manifest_id=manifest_id,
                master_type=master_type,
                operation_id=operation_id,
                idempotency_key=idempotency_key,
                phase="PREPARED",
                created_at=now,
                updated_at=now,
                target_files=targets,
                backup_files=backup_files,
                content_hashes=hashes,
                metadata={"originally_missing": originally_missing},
            )
            self._write_manifest(manifest)
            manifest = manifest.model_copy(
                update={"phase": "REPLACING", "updated_at": now}
            )
            self._write_manifest(manifest)

            for target in targets:
                staged_path = self._path(f"{staging_base}/new/{target}")
                target_path = self._path(target)
                target_path.parent.mkdir(parents=True, exist_ok=True)
                os.replace(staged_path, target_path)

            manifest = manifest.model_copy(
                update={"phase": "COMMITTED", "updated_at": now}
            )
            self._write_manifest(manifest)
            shutil.rmtree(self._path(staging_base), ignore_errors=True)
            shutil.rmtree(self._path(recovery_base), ignore_errors=True)
            return manifest_id
        except Exception as exc:
            if "manifest" not in locals():
                shutil.rmtree(self._path(staging_base), ignore_errors=True)
                shutil.rmtree(self._path(recovery_base), ignore_errors=True)
                raise LocalMasterAtomicWriteError(
                    self._exception_message("transaction preparation")
                ) from exc
            try:
                self._restore_manifest(manifest)
                rolled_back = manifest.model_copy(
                    update={"phase": "ROLLED_BACK", "updated_at": now}
                )
                self._write_manifest(rolled_back)
                shutil.rmtree(self._path(staging_base), ignore_errors=True)
                shutil.rmtree(self._path(recovery_base), ignore_errors=True)
            except Exception as recovery_exc:
                try:
                    required = manifest.model_copy(
                        update={"phase": "RECOVERY_REQUIRED", "updated_at": now}
                    )
                    self._write_manifest(required)
                except Exception:
                    pass
                raise LocalMasterRecoveryError(
                    "transaction failed and rollback needs recovery"
                ) from recovery_exc
            raise LocalMasterAtomicWriteError(
                self._exception_message("transaction")
            ) from exc

    def initialize(
        self,
        *,
        initialized_at: datetime,
        initialized_by: ActorSystem,
        operation_id: UUID,
    ) -> None:
        if initialized_at.tzinfo is None or initialized_at.utcoffset() != timedelta(0):
            raise LocalMasterInitializationError(
                "initialization time must be timezone-aware UTC"
            )
        required = list(_STATE_FILES)
        existing = [self._path(item).exists() for item in required]
        if any(existing):
            if not all(existing):
                raise LocalMasterInitializationError(
                    "existing local master state is incomplete"
                )
            try:
                for master in (MasterType.ACCOUNTING, MasterType.BANKING):
                    self._verify_integrity(master)
                self.read_audit_events()
                self.read_status_history()
            except Exception as exc:
                raise LocalMasterInitializationError(
                    "existing local master state is invalid"
                ) from exc
            return

        empty_hash = self._hash_value([])
        files = {
            "accounting/rows.json": b"[]",
            "banking/rows.json": b"[]",
            "audit/events.jsonl": b"",
            "audit/status_history.jsonl": b"",
        }
        for master in (MasterType.ACCOUNTING, MasterType.BANKING):
            version = MasterVersion(
                master_name=master,
                current_version=0,
                content_hash=empty_hash,
                last_operation_id=operation_id,
                updated_at=initialized_at,
                updated_by=initialized_by,
            )
            files[f"{self._master_name(master)}/master_version.json"] = (
                self._canonical_bytes(self._model_json(version))
            )
        try:
            self._transaction(
                master_type=MasterType.ACCOUNTING,
                operation_id=operation_id,
                idempotency_key=f"initialize:{operation_id}",
                now=initialized_at,
                files=files,
            )
        except Exception as exc:
            raise LocalMasterInitializationError(
                "local master initialization failed"
            ) from exc

    def read_master_version(self, master_type: MasterType) -> MasterVersion:
        version = self._read_model(
            self._relative(master_type, "master_version.json"),
            MasterVersion,
        )
        if version.master_name != master_type:
            raise LocalMasterCorruptionError("master version has an invalid master type")
        return version

    def read_accounting_rows(
        self,
        *,
        include_deleted: bool = False,
    ) -> list[AccountingDocumentMasterRow]:
        return self._read_rows(MasterType.ACCOUNTING, include_deleted)  # type: ignore[return-value]

    def read_banking_rows(
        self,
        *,
        include_deleted: bool = False,
    ) -> list[BankTransactionMasterRow]:
        return self._read_rows(MasterType.BANKING, include_deleted)  # type: ignore[return-value]

    def read_audit_events(self) -> list[AuditEvent]:
        return self._read_jsonl("audit/events.jsonl", AuditEvent)

    def read_status_history(self) -> list[StatusHistory]:
        return self._read_jsonl("audit/status_history.jsonl", StatusHistory)

    def read_lock(self, master_type: MasterType) -> ResourceLock | None:
        relative = self._relative(master_type, "lock.json")
        if not self._path(relative).exists():
            return None
        return self._read_model(relative, ResourceLock)

    def _append_files(
        self,
        *,
        master_type: MasterType,
        operation_id: UUID,
        idempotency_key: str,
        now: datetime,
        files: dict[str, bytes],
    ) -> None:
        self._transaction(
            master_type=master_type,
            operation_id=operation_id,
            idempotency_key=idempotency_key,
            now=now,
            files=files,
        )

    @staticmethod
    def _jsonl_bytes(events: list[BaseModel]) -> bytes:
        return b"".join(
            json.dumps(
                event.model_dump(mode="json"),
                sort_keys=True,
                ensure_ascii=False,
                separators=(",", ":"),
            ).encode("utf-8")
            + b"\n"
            for event in events
        )

    def _record_lock_event(
        self,
        master_type: MasterType,
        action: AuditAction,
        *,
        operation_id: UUID,
        occurred_at: datetime,
        lock: ResourceLock,
    ) -> AuditEvent:
        return AuditEvent(
            event_id=uuid4(),
            occurred_at=occurred_at,
            operation_id=operation_id,
            resource_type=AuditResourceType.LOCK,
            resource_id=lock.resource_name,
            action=action,
            actor_system=lock.owner_system,
            actor_id=lock.owner_instance_id,
            previous_version=lock.base_version,
            new_version=lock.base_version,
            reason=f"{master_type.value} lock {action.value.lower()}",
        )

    def acquire_lock(
        self,
        master_type: MasterType,
        *,
        owner_system: ActorSystem,
        owner_instance_id: str,
        operation_id: UUID,
        acquired_at: datetime,
        expires_at: datetime,
    ) -> ResourceLock:
        current = self.read_master_version(master_type)
        existing = self.read_lock(master_type)
        if not can_acquire_lock(existing, now=acquired_at):
            raise LockUnavailableError("master resource is already locked")
        lock = ResourceLock(
            lock_id=uuid4(),
            resource_name=master_type.value,
            owner_system=owner_system,
            owner_instance_id=owner_instance_id,
            operation_id=operation_id,
            acquired_at=acquired_at,
            heartbeat_at=acquired_at,
            expires_at=expires_at,
            base_version=current.current_version,
            status=LockStatus.ACTIVE,
        )
        event = self._record_lock_event(
            master_type,
            AuditAction.LOCK_ACQUIRED,
            operation_id=operation_id,
            occurred_at=acquired_at,
            lock=lock,
        )
        events = self.read_audit_events()
        if existing is not None and existing.status == LockStatus.ACTIVE:
            # can_acquire_lock only allows an ACTIVE lock here when it has expired.
            events.append(
                self._record_lock_event(
                    master_type,
                    AuditAction.LOCK_EXPIRED,
                    operation_id=existing.operation_id,
                    occurred_at=acquired_at,
                    lock=existing,
                )
            )
        events.append(event)
        self._append_files(
            master_type=master_type,
            operation_id=operation_id,
            idempotency_key=f"lock-acquire:{lock.lock_id}",
            now=acquired_at,
            files={
                self._relative(master_type, "lock.json"): self._canonical_bytes(
                    self._model_json(lock)
                ),
                "audit/events.jsonl": self._jsonl_bytes(events),
            },
        )
        return lock

    def _owned_lock(
        self,
        master_type: MasterType,
        *,
        lock_id: UUID,
        owner_system: ActorSystem,
        owner_instance_id: str,
        now: datetime,
    ) -> ResourceLock:
        lock = self.read_lock(master_type)
        if lock is None:
            raise LockUnavailableError("master resource has no lock")
        if lock.lock_id != lock_id:
            raise LockOwnershipError("lock identifier does not match")
        if lock.owner_system != owner_system or lock.owner_instance_id != owner_instance_id:
            raise LockOwnershipError("caller does not own the lock")
        if lock.is_expired(now):
            raise LockExpiredError("master resource lock has expired")
        if lock.status != LockStatus.ACTIVE:
            raise LockUnavailableError("master resource lock is not active")
        if now < lock.heartbeat_at:
            raise MasterTableError("operation time precedes the lock heartbeat")
        return lock

    @staticmethod
    def _released_lock(lock: ResourceLock, released_at: datetime) -> ResourceLock:
        values = lock.model_dump()
        values.update(
            status=LockStatus.RELEASED,
            heartbeat_at=released_at,
            expires_at=max(lock.expires_at, released_at + timedelta(microseconds=1)),
        )
        return ResourceLock.model_validate(values)

    def renew_lock(
        self,
        master_type: MasterType,
        *,
        lock_id: UUID,
        owner_system: ActorSystem,
        owner_instance_id: str,
        now: datetime,
        new_expires_at: datetime,
    ) -> ResourceLock:
        old = self._owned_lock(
            master_type,
            lock_id=lock_id,
            owner_system=owner_system,
            owner_instance_id=owner_instance_id,
            now=now,
        )
        renewed = old.renew(now, new_expires_at)
        event = self._record_lock_event(
            master_type,
            AuditAction.LOCK_RENEWED,
            operation_id=old.operation_id,
            occurred_at=now,
            lock=renewed,
        )
        events = self.read_audit_events() + [event]
        self._append_files(
            master_type=master_type,
            operation_id=old.operation_id,
            idempotency_key=f"lock-renew:{lock_id}:{now.isoformat()}",
            now=now,
            files={
                self._relative(master_type, "lock.json"): self._canonical_bytes(
                    self._model_json(renewed)
                ),
                "audit/events.jsonl": self._jsonl_bytes(events),
            },
        )
        return renewed

    def release_lock(
        self,
        master_type: MasterType,
        *,
        lock_id: UUID,
        owner_system: ActorSystem,
        owner_instance_id: str,
        released_at: datetime,
    ) -> None:
        old = self._owned_lock(
            master_type,
            lock_id=lock_id,
            owner_system=owner_system,
            owner_instance_id=owner_instance_id,
            now=released_at,
        )
        released = self._released_lock(old, released_at)
        event = self._record_lock_event(
            master_type,
            AuditAction.LOCK_RELEASED,
            operation_id=old.operation_id,
            occurred_at=released_at,
            lock=released,
        )
        events = self.read_audit_events() + [event]
        self._append_files(
            master_type=master_type,
            operation_id=old.operation_id,
            idempotency_key=f"lock-release:{lock_id}",
            now=released_at,
            files={
                self._relative(master_type, "lock.json"): self._canonical_bytes(
                    self._model_json(released)
                ),
                "audit/events.jsonl": self._jsonl_bytes(events),
            },
        )

    @staticmethod
    def _row_id(master_type: MasterType, row: BaseModel) -> UUID:
        return row.document_id if master_type == MasterType.ACCOUNTING else row.transaction_id

    @staticmethod
    def _status_events(
        master_type: MasterType,
        old: BaseModel | None,
        new: BaseModel,
        *,
        operation_id: UUID,
        changed_at: datetime,
    ) -> list[StatusHistory]:
        if old is None:
            return []
        specs = (
            _ACCOUNTING_STATUSES
            if master_type == MasterType.ACCOUNTING
            else _BANKING_STATUSES
        )
        result = []
        for name, domain in specs:
            old_value: Any = old
            new_value: Any = new
            if "." in name:
                first, second = name.split(".", 1)
                old_value, new_value = getattr(old_value, first), getattr(new_value, first)
                old_status, new_status = getattr(old_value, second), getattr(new_value, second)
            else:
                old_status, new_status = getattr(old_value, name), getattr(new_value, name)
            if old_status == new_status:
                continue
            result.append(
                StatusHistory(
                    status_history_id=uuid4(),
                    resource_type=(
                        AuditResourceType.ACCOUNTING_DOCUMENT
                        if master_type == MasterType.ACCOUNTING
                        else AuditResourceType.BANK_TRANSACTION
                    ),
                    resource_id=str(LocalMasterWriter._row_id(master_type, new)),
                    status_domain=domain,
                    status_from=str(old_status.value),
                    status_to=str(new_status.value),
                    changed_at=changed_at,
                    changed_by=new.updated_by,
                    operation_id=operation_id,
                    row_version_after_change=new.row_version,
                )
            )
        if master_type == MasterType.BANKING:
            old_batch = old.banking_record.payment_batch
            new_batch = new.banking_record.payment_batch
            old_status = old_batch.status if old_batch is not None else None
            new_status = new_batch.status if new_batch is not None else None
            if old_status != new_status and new_status is not None:
                result.append(
                    StatusHistory(
                        status_history_id=uuid4(),
                        resource_type=AuditResourceType.PAYMENT_BATCH,
                        resource_id=str(new_batch.batch_id),
                        status_domain=StatusDomain.RECONCILIATION,
                        status_from=old_status.value if old_status is not None else None,
                        status_to=new_status.value,
                        changed_at=changed_at,
                        changed_by=new.updated_by,
                        operation_id=operation_id,
                        row_version_after_change=new.row_version,
                    )
                )
        return result

    @staticmethod
    def _changed_fields(old: BaseModel, new: BaseModel) -> list[str]:
        old_data = old.model_dump(mode="json")
        new_data = new.model_dump(mode="json")
        return sorted(
            key
            for key in old_data.keys() | new_data.keys()
            if old_data.get(key) != new_data.get(key)
        )

    def _publish(
        self,
        master_type: MasterType,
        rows: list[BaseModel],
        *,
        lock_id: UUID,
        owner_system: ActorSystem,
        owner_instance_id: str,
        expected_base_version: int,
        operation_id: UUID,
        idempotency_key: str,
        committed_at: datetime,
        audit_events: list[AuditEvent] | None,
        status_history: list[StatusHistory] | None,
    ) -> LocalMasterWriteResult:
        """Publish rows, audit, status history and the version in one local transaction.

        Decision for publications containing only unchanged rows (strategy A):
        the publication is still a write. It creates a new MasterVersion, consumes
        the idempotency_key, releases the lock and appends its audit event, and the
        result lists the rows under unchanged_row_ids. No row is rewritten with new
        content. This keeps LocalMasterWriteResult consistent (new version is always
        previous + 1) and makes every acknowledged operation replay-detectable.

        The transaction is recoverable, not ACID: os.replace is atomic per file only,
        and multi-file consistency relies on the manifest, backups and recovery.
        """
        if committed_at.tzinfo is None or committed_at.utcoffset() != timedelta(0):
            raise MasterTableError("committed_at must be timezone-aware UTC")
        if not rows and not audit_events:
            raise MasterTableError("publication requires rows or explicit audit events")

        lock = self._owned_lock(
            master_type,
            lock_id=lock_id,
            owner_system=owner_system,
            owner_instance_id=owner_instance_id,
            now=committed_at,
        )
        current_version = self.read_master_version(master_type)
        validate_write_preconditions(
            lock=lock,
            now=committed_at,
            owner_system=owner_system,
            owner_instance_id=owner_instance_id,
            expected_base_version=expected_base_version,
            master_version=current_version,
            idempotency_key=idempotency_key,
        )

        model_type: type[BaseModel] = (
            AccountingDocumentMasterRow
            if master_type == MasterType.ACCOUNTING
            else BankTransactionMasterRow
        )
        existing_rows = self._read_rows(master_type, include_deleted=True)
        self._verify_rows_hash(master_type, existing_rows, current_version)
        existing_by_id = {self._row_id(master_type, row): row for row in existing_rows}
        supplied_ids: set[UUID] = set()
        created_ids: list[UUID] = []
        updated_ids: list[UUID] = []
        unchanged_ids: list[UUID] = []
        output_by_id = dict(existing_by_id)
        generated_events: list[AuditEvent] = []
        generated_history: list[StatusHistory] = []

        for supplied in rows:
            try:
                row = model_type.model_validate(supplied.model_dump(mode="json"))
            except Exception as exc:
                raise LocalMasterRowConflictError("published row failed validation") from exc
            self._ensure_no_absolute_paths(row.model_dump(mode="json"))
            row_id = self._row_id(master_type, row)
            if row_id in supplied_ids:
                raise LocalMasterRowConflictError("duplicate row identifier in publication")
            supplied_ids.add(row_id)
            previous = existing_by_id.get(row_id)
            if previous is None:
                if row.row_version != 1:
                    raise LocalMasterRowConflictError(
                        "new rows must start at row_version 1"
                    )
                output_by_id[row_id] = row
                created_ids.append(row_id)
                action = (
                    AuditAction.IMPORTED
                    if (
                        row.ingestion_status.value == "IMPORTED"
                        if master_type == MasterType.ACCOUNTING
                        else row.import_status.value == "IMPORTED"
                    )
                    else AuditAction.CREATED
                )
                generated_events.append(
                    self._row_audit_event(
                        master_type,
                        row,
                        operation_id=operation_id,
                        committed_at=committed_at,
                        action=action,
                        previous=None,
                    )
                )
            elif row.row_version == previous.row_version:
                if row.content_hash != previous.content_hash:
                    raise LocalMasterRowConflictError(
                        "same row version has a different content hash"
                    )
                if row.model_dump(mode="json") != previous.model_dump(mode="json"):
                    raise LocalMasterRowConflictError(
                        "same row version and content hash with a different payload"
                    )
                unchanged_ids.append(row_id)
                continue
            elif row.row_version < previous.row_version:
                raise LocalMasterRowConflictError("row version regression")
            elif row.row_version != previous.row_version + 1:
                raise LocalMasterRowConflictError("row version must increment by one")
            else:
                if row.content_hash == previous.content_hash:
                    raise LocalMasterRowConflictError(
                        "a new row version must change content_hash"
                    )
                if (
                    row.created_at != previous.created_at
                    or row.created_by != previous.created_by
                ):
                    raise LocalMasterRowConflictError(
                        "row creation metadata is immutable"
                    )
                if row.updated_at < previous.updated_at:
                    raise LocalMasterRowConflictError(
                        "updated_at cannot move backwards"
                    )
                output_by_id[row_id] = row
                updated_ids.append(row_id)
                action = (
                    AuditAction.RESTORED
                    if previous.is_deleted and not row.is_deleted
                    else AuditAction.SOFT_DELETED
                    if not previous.is_deleted and row.is_deleted
                    else AuditAction.UPDATED
                )
                generated_events.append(
                    self._row_audit_event(
                        master_type,
                        row,
                        operation_id=operation_id,
                        committed_at=committed_at,
                        action=action,
                        previous=previous,
                    )
                )
                generated_history.extend(
                    self._status_events(
                        master_type,
                        previous,
                        row,
                        operation_id=operation_id,
                        changed_at=committed_at,
                    )
                )

        existing_events = self.read_audit_events()
        existing_history = self.read_status_history()
        event_ids = {event.event_id for event in existing_events}
        history_ids = {item.status_history_id for item in existing_history}
        supplied_events = list(audit_events or [])
        supplied_history = list(status_history or [])
        for event in supplied_events:
            try:
                validated = AuditEvent.model_validate(event.model_dump(mode="json"))
            except Exception as exc:
                raise LocalMasterRowConflictError("supplied audit event is invalid") from exc
            if validated.event_id in event_ids:
                raise LocalMasterRowConflictError("duplicate audit event identifier")
            event_ids.add(validated.event_id)
            self._ensure_no_absolute_paths(validated.model_dump(mode="json"))
        for item in supplied_history:
            try:
                validated = StatusHistory.model_validate(item.model_dump(mode="json"))
            except Exception as exc:
                raise LocalMasterRowConflictError(
                    "supplied status history is invalid"
                ) from exc
            if validated.status_history_id in history_ids:
                raise LocalMasterRowConflictError("duplicate status history identifier")
            history_ids.add(validated.status_history_id)
            self._ensure_no_absolute_paths(validated.model_dump(mode="json"))

        all_new_events = generated_events + supplied_events
        all_new_history = generated_history + supplied_history
        new_rows = sorted(output_by_id.values(), key=lambda row: str(self._row_id(master_type, row)))
        row_payloads = [row.model_dump(mode="json") for row in new_rows]
        state_hash = self._hash_value(row_payloads)
        next_version = next_master_version(
            current_version,
            operation_id=operation_id,
            actor=owner_system,
            idempotency_key=idempotency_key,
            content_hash=state_hash,
            updated_at=committed_at,
        )
        released_lock = self._released_lock(lock, committed_at)
        release_event = self._record_lock_event(
            master_type,
            AuditAction.LOCK_RELEASED,
            operation_id=operation_id,
            occurred_at=committed_at,
            lock=released_lock,
        )
        all_new_events.append(release_event)
        combined_events = existing_events + all_new_events
        combined_history = existing_history + all_new_history

        targets = {
            self._relative(master_type, "rows.json"): self._canonical_bytes(row_payloads),
            self._relative(master_type, "master_version.json"): self._canonical_bytes(
                self._model_json(next_version)
            ),
            self._relative(master_type, "lock.json"): self._canonical_bytes(
                self._model_json(released_lock)
            ),
            "audit/events.jsonl": self._jsonl_bytes(combined_events),
            "audit/status_history.jsonl": self._jsonl_bytes(combined_history),
        }
        self._ensure_no_absolute_paths(row_payloads)
        try:
            result = LocalMasterWriteResult(
                master_type=master_type,
                operation_id=operation_id,
                previous_master_version=current_version.current_version,
                new_master_version=next_version.current_version,
                created_row_ids=created_ids,
                updated_row_ids=updated_ids,
                unchanged_row_ids=unchanged_ids,
                audit_event_ids=[event.event_id for event in all_new_events],
                status_history_ids=[item.status_history_id for item in all_new_history],
                committed_at=committed_at,
                content_hash=state_hash,
                idempotency_key=idempotency_key,
            )
        except ValidationError as exc:
            raise LocalMasterRowConflictError(
                "publication result identifiers or versions are inconsistent"
            ) from exc
        self._append_files(
            master_type=master_type,
            operation_id=operation_id,
            idempotency_key=idempotency_key,
            now=committed_at,
            files=targets,
        )
        return result

    @staticmethod

            occurred_at=committed_at,
            operation_id=operation_id,
            resource_type=(
                AuditResourceType.ACCOUNTING_DOCUMENT
                if master_type == MasterType.ACCOUNTING

            action=action,
            actor_system=(
                row.updated_by if master_type == MasterType.ACCOUNTING else row.updated_by

            previous_version=previous.row_version if previous is not None else 0,
            new_version=row.row_version,
            before_hash=previous.content_hash if previous is not None else None,
            after_hash=row.content_hash,
            changed_fields=(
                LocalMasterWriter._changed_fields(previous, row)
                if previous is not None
                else sorted(row.model_dump(mode="json").keys())
            ),
        )

    def _verify_recovered_state(self) -> bool:
        """Validates the restored state; returns False when no state file exists."""
        present = [self._path(name).exists() for name in _STATE_FILES]
        if not any(present):
            return False
        if not all(present):
            raise LocalMasterRecoveryError("local master state is incomplete after recovery")
        for master in (MasterType.ACCOUNTING, MasterType.BANKING):
            self._verify_integrity(master)
        self.read_audit_events()
        self.read_status_history()
        return True

    def recover(
        self,
        *,
        recovered_at: datetime,
        recovered_by: ActorSystem,
    ) -> list[UUID]:
        if recovered_at.tzinfo is None or recovered_at.utcoffset() != timedelta(0):
            raise LocalMasterRecoveryError("recovery timestamp must be UTC")
        recovery_dir = self._path("recovery")
        if not recovery_dir.exists():
            return []
        recovered: list[UUID] = []
        for manifest_path in sorted(recovery_dir.glob("*.json")):
            try:
                manifest = LocalTransactionManifest.model_validate_json(
                    manifest_path.read_bytes()
                )
            except Exception as exc:
                raise LocalMasterRecoveryError(
                    "transaction manifest is corrupt"
                ) from exc
            if manifest.phase in _RESOLVED_PHASES:
                continue
            try:
                self._restore_manifest(manifest)
                initialized = self._verify_recovered_state()

                events = self.read_audit_events() if initialized else []
                if initialized and not any(
                    event.metadata.get("recovered_manifest_id")
                    == str(manifest.transaction_manifest_id)
                    for event in events
                ):
                    events.append(
                        AuditEvent(
                            event_id=uuid4(),
                            occurred_at=recovered_at,
                            operation_id=manifest.operation_id,
                            resource_type=AuditResourceType.MASTER_TABLE,
                            resource_id=manifest.master_type.value,
                            action=AuditAction.WRITE_REJECTED,
                            actor_system=recovered_by,
                            reason="Recovered incomplete local transaction",
                            metadata={
                                "recovered_manifest_id": str(
                                    manifest.transaction_manifest_id
                                )
                            },
                        )
                    )
                    self._atomic_write(
                        self._path("audit/events.jsonl"),
                        self._jsonl_bytes(events),
                    )
                rolled_back = manifest.model_copy(
                    update={"phase": "ROLLED_BACK", "updated_at": recovered_at}
                )
                self._write_manifest(rolled_back)
                shutil.rmtree(
                    self._path(f"staging/{manifest.transaction_manifest_id}"),
                    ignore_errors=True,
                )
                shutil.rmtree(
                    self._path(f"recovery/{manifest.transaction_manifest_id}"),
                    ignore_errors=True,
                )
                recovered.append(manifest.transaction_manifest_id)
            except Exception as exc:
                try:
                    self._write_manifest(
                        manifest.model_copy(
                            update={"phase": "RECOVERY_REQUIRED", "updated_at": recovered_at}
                        )
                    )
                except Exception:
                    pass
                raise LocalMasterRecoveryError(
                    "local transaction recovery failed"
                ) from exc
        return recovered

def acquire(
    writer: LocalMasterWriter,
    master_type: MasterType = MasterType.ACCOUNTING,
    *,
    owner: ActorSystem = ActorSystem.LOCAL,
) -> Any:
    return writer.acquire_lock(
        master_type,
        owner_system=owner,
        owner_instance_id=f"{owner.value.lower()}-instance",
        operation_id=uuid4(),
        acquired_at=LATER,
        expires_at=LATER + timedelta(hours=1),
    )


def publish_accounting(
    writer: LocalMasterWriter,
    rows: list[AccountingDocumentMasterRow],
    *,
    lock: Any | None = None,
    expected_version: int = 0,
    operation_id: UUID = OPERATION_ID,
    idempotency_key: str = "publish-accounting-1",
    committed_at: datetime = LATER + timedelta(minutes=1),
    audit_events: list[AuditEvent] | None = None,
    status_history: list[StatusHistory] | None = None,
) -> LocalMasterWriteResult:
    held = lock or acquire(writer)
    return writer.publish_accounting_rows(
        rows,
        lock_id=held.lock_id,
        owner_system=held.owner_system,
        owner_instance_id=held.owner_instance_id,
        expected_base_version=expected_version,
        operation_id=operation_id,
        idempotency_key=idempotency_key,
        committed_at=committed_at,
        audit_events=audit_events,
        status_history=status_history,
    )

def test_pre_replace_failure_leaves_state_unchanged(
    initialized_writer: LocalMasterWriter,
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    lock = acquire(initialized_writer)
    before = (initialized_writer._root / "accounting" / "rows.json").read_bytes()
    original_replace = writer_module.os.replace
    failed = False

    def fail_replace(source: Any, target: Any) -> None:
        nonlocal failed
        if not failed and Path(target).name == "rows.json":
            failed = True
            raise OSError("simulated failure")
        original_replace(source, target)

    monkeypatch.setattr(writer_module.os, "replace", fail_replace)
    with pytest.raises(LocalMasterAtomicWriteError):

def test_preparation_failure_cleans_staging_and_backup_directories(
    initialized_writer: LocalMasterWriter,
    monkeypatch: pytest.MonkeyPatch,
) -> None:

    original_write = initialized_writer._write_fsynced

        item
        for item in (initialized_writer._root / "recovery").iterdir()
        if item.is_dir()
    ]
    assert recovery_dirs == []


def test_failure_during_replace_rolls_back_all_final_files(
    initialized_writer: LocalMasterWriter,
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    lock = acquire(initialized_writer)
    targets = [
        "accounting/rows.json",
        "accounting/master_version.json",
        "accounting/lock.json",
        "audit/events.jsonl",
        "audit/status_history.jsonl",
    ]
    before = {
        target: (initialized_writer._root / target).read_bytes()
        for target in targets
        if (initialized_writer._root / target).exists()
    }
    before_audit_events = initialized_writer.read_audit_events()
    original_replace = writer_module.os.replace
    failed = False

    def fail_once(source: Any, target: Any) -> None:
        nonlocal failed
        if not failed and Path(target).name == "master_version.json":
            failed = True
            raise OSError("simulated replace failure")
        original_replace(source, target)

    monkeypatch.setattr(writer_module.os, "replace", fail_once)
    with pytest.raises(LocalMasterAtomicWriteError):
        publish_accounting(initialized_writer, [accounting_row()], lock=lock)
    monkeypatch.setattr(writer_module.os, "replace", original_replace)

    for target, contents in before.items():
        assert (initialized_writer._root / target).read_bytes() == contents
    assert initialized_writer.read_audit_events() == before_audit_events

def snapshot(root: Path) -> dict[str, bytes]:
    return {
        path.relative_to(root).as_posix(): path.read_bytes()
        for path in sorted(root.rglob("*"))
        if path.is_file()
    }


def state_snapshot(root: Path) -> dict[str, bytes]:
    return {
        name: data
        for name, data in snapshot(root).items()
        if name.split("/")[0] in {"accounting", "banking", "audit"}
    }


def manifest_phases(root: Path) -> list[str]:
    return sorted(
        LocalTransactionManifest.model_validate_json(path.read_bytes()).phase
        for path in (root / "recovery").glob("*.json")
    )


def exception_chain(error: BaseException) -> list[BaseException]:
    chain: list[BaseException] = []
    current: BaseException | None = error
    while current is not None and current not in chain:
        chain.append(current)
        current = current.__cause__ or current.__context__
    return chain


def fail_replace_at(
    monkeypatch: pytest.MonkeyPatch,
    filename: str,
    before_fail: Any = None,
) -> None:
    original = writer_module.os.replace
    state = {"failed": False}

    def hook(source: Any, target: Any) -> None:
        if not state["failed"] and Path(target).name == filename:
            state["failed"] = True
            if before_fail is not None:
                before_fail()
            raise OSError("simulated replace failure")
        original(source, target)

    monkeypatch.setattr(writer_module.os, "replace", hook)

def test_bank_row_update_n_to_n_plus_one_preserves_other_row_and_records_status(
    initialized_writer: LocalMasterWriter,
) -> None:
    first = bank_master(idempotency_key="bank-1")
    second = bank_master(
        transaction_id=TRANSACTION_ID_2,
        idempotency_key="bank-2",
    )
    publish_banking(initialized_writer, [first, second])
    other_before = next(
        row for row in initialized_writer.read_banking_rows() if row.transaction_id == TRANSACTION_ID_2
    ).model_dump(mode="json")

    updated = bank_master(
        row_version=2,
        content_hash=HASH_B,
        updated_at=COMMIT_1,
        record_overrides={"reconciliation_status": ReconciliationStatus.NEEDS_REVIEW},
    )
    result = publish_banking(
        initialized_writer,
        [updated],
        expected_version=1,
        idempotency_key="bank-update",
    )

    by_id = {row.transaction_id: row for row in initialized_writer.read_banking_rows()}
    assert result.updated_row_ids == [TRANSACTION_ID]
    assert by_id[TRANSACTION_ID].row_version == 2
    assert by_id[TRANSACTION_ID_2].model_dump(mode="json") == other_before
    history = initialized_writer.read_status_history()
    assert len(history) == 1
    assert history[0].resource_type == AuditResourceType.BANK_TRANSACTION
    assert history[0].status_domain == StatusDomain.RECONCILIATION
    assert (history[0].status_from, history[0].status_to) == ("UNMATCHED", "NEEDS_REVIEW")

def test_payment_batch_status_change_is_recorded_in_history(
    initialized_writer: LocalMasterWriter,
) -> None:
    batch_id = uuid4()
    amounts = ["1000.00", "2000.00"]
    open_batch = payment_batch(PaymentBatchType.GROUPED_TRANSFER, amounts, batch_id=batch_id)
    rejected = payment_batch(
        PaymentBatchType.GROUPED_TRANSFER,
        amounts,
        batch_id=batch_id,
        status=PaymentBatchStatus.REJECTED,
    )

    def row(batch: PaymentBatch, **overrides: Any) -> BankTransactionMasterRow:
        return bank_master(
            record_overrides={
                "amount": batch.total_amount,
                "unallocated_amount": batch.total_amount,
                "payment_batch": batch,
            },
            **overrides,
        )

    publish_banking(initialized_writer, [row(open_batch)])
    publish_banking(
        initialized_writer,
        [row(rejected, row_version=2, content_hash=HASH_B, updated_at=COMMIT_1)],
        expected_version=1,
        idempotency_key="batch-rejected",
    )

    history = initialized_writer.read_status_history()
    assert len(history) == 1
    assert history[0].resource_type == AuditResourceType.PAYMENT_BATCH
    assert history[0].resource_id == str(batch_id)
    assert (history[0].status_from, history[0].status_to) == ("NEEDS_REVIEW", "REJECTED")

@pytest.mark.parametrize("master_type", [MasterType.ACCOUNTING, MasterType.BANKING])

    initialized_writer: LocalMasterWriter,
    master_type: MasterType,
) -> None:
    before = initialized_writer.read_master_version(master_type)
    held = acquire(initialized_writer, master_type)

        owner_instance_id=held.owner_instance_id,

        new_expires_at=renewed_at + timedelta(hours=1),

    initialized_writer.release_lock(
        master_type,
        lock_id=held.lock_id,
        owner_system=held.owner_system,
        owner_instance_id=held.owner_instance_id,
        released_at=renewed_at,
    )

    assert initialized_writer.read_master_version(master_type) == before

def test_unchanged_only_publication_creates_a_version_and_consumes_the_key(
    initialized_writer: LocalMasterWriter,
) -> None:
    row = accounting_row()
    publish_accounting(initialized_writer, [row])
    row_events_before = [
        event
        for event in initialized_writer.read_audit_events()
        if event.resource_type == AuditResourceType.ACCOUNTING_DOCUMENT
    ]
    rows_bytes = (initialized_writer._root / "accounting" / "rows.json").read_bytes()

    result = publish_accounting(
        initialized_writer,
        [row],
        expected_version=1,
        idempotency_key="no-op-key",
    )

    version = initialized_writer.read_master_version(MasterType.ACCOUNTING)
    assert result.unchanged_row_ids == [DOCUMENT_ID]
    assert result.created_row_ids == [] and result.updated_row_ids == []
    assert (result.previous_master_version, result.new_master_version) == (1, 2)
    assert version.current_version == 2
    assert "no-op-key" in version.applied_idempotency_keys
    assert (initialized_writer._root / "accounting" / "rows.json").read_bytes() == rows_bytes
    assert [
        event
        for event in initialized_writer.read_audit_events()
        if event.resource_type == AuditResourceType.ACCOUNTING_DOCUMENT
    ] == row_events_before
    assert initialized_writer.read_status_history() == []
    assert initialized_writer.read_lock(MasterType.ACCOUNTING).status == LockStatus.RELEASED

    with pytest.raises(DuplicateOperationError):
        publish_accounting(
            initialized_writer,
            [row],
            expected_version=2,
            idempotency_key="no-op-key",
        )

def test_failure_after_rows_replaced_but_before_master_version_restores_everything(
    initialized_writer: LocalMasterWriter,
    monkeypatch: pytest.MonkeyPatch,
    tmp_path: Path,
) -> None:
    lock = acquire(initialized_writer)
    before = state_snapshot(tmp_path)
    seen: dict[str, bool] = {}

    def observe_partial_state() -> None:
        seen["rows_replaced"] = (tmp_path / "accounting" / "rows.json").read_bytes() != before[
            "accounting/rows.json"
        ]
        seen["version_untouched"] = (
            tmp_path / "accounting" / "master_version.json"
        ).read_bytes() == before["accounting/master_version.json"]

    fail_replace_at(monkeypatch, "master_version.json", observe_partial_state)
    with pytest.raises(LocalMasterAtomicWriteError):
        publish_accounting(
            initialized_writer,
            [accounting_row()],
            lock=lock,
            status_history=[make_status_history()],
            audit_events=[make_audit_event()],
        )

    assert seen == {"rows_replaced": True, "version_untouched": True}
    assert state_snapshot(tmp_path) == before
    assert initialized_writer.read_lock(MasterType.ACCOUNTING).lock_id == lock.lock_id
    assert initialized_writer.read_lock(MasterType.ACCOUNTING).status == LockStatus.ACTIVE
    assert initialized_writer.read_master_version(MasterType.ACCOUNTING).current_version == 0
    assert initialized_writer.read_status_history() == []
    assert len(initialized_writer.read_audit_events()) == 1  # lock acquisition only

    # The same lock and idempotency key remain usable after a clean rollback.
    result = publish_accounting(initialized_writer, [accounting_row()], lock=lock)
    assert result.new_master_version == 1


@pytest.mark.parametrize(

def test_recover_restores_prepared_and_replacing_manifests_idempotently(
    initialized_writer: LocalMasterWriter,
    tmp_path: Path,
) -> None:
    prepared = write_interrupted_manifest(tmp_path, "PREPARED")
    replacing = write_interrupted_manifest(tmp_path, "REPLACING")

    recovered = initialized_writer.recover(recovered_at=LATER, recovered_by=ActorSystem.LOCAL)

    assert sorted(recovered) == sorted([prepared, replacing])
    assert initialized_writer.read_accounting_rows() == []
    for manifest_id in (prepared, replacing):
        assert read_manifest(tmp_path, manifest_id).phase == "ROLLED_BACK"
        assert not (tmp_path / "recovery" / str(manifest_id)).exists()
    events_after_first = initialized_writer.read_audit_events()
    assert initialized_writer.recover(recovered_at=LATER, recovered_by=ActorSystem.LOCAL) == []
    assert initialized_writer.read_audit_events() == events_after_first
    recovered_ids = [event.metadata["recovered_manifest_id"] for event in events_after_first]
    assert sorted(recovered_ids) == sorted([str(prepared), str(replacing)])


def test_recover_with_missing_backup_marks_recovery_required_and_keeps_evidence(
    initialized_writer: LocalMasterWriter,
    tmp_path: Path,
) -> None:
    manifest_id = uuid4()
    backup_dir = tmp_path / "recovery" / str(manifest_id) / "backup"
    (backup_dir / "accounting").mkdir(parents=True)
    kept = backup_dir / "accounting" / "rows.json"
    kept.write_bytes(b"[]")
    manifest = LocalTransactionManifest(
        transaction_manifest_id=manifest_id,
        master_type=MasterType.ACCOUNTING,
        operation_id=uuid4(),
        idempotency_key="missing-backup",
        phase="REPLACING",
        created_at=NOW,
        updated_at=NOW,
        target_files=["accounting/rows.json", "banking/rows.json"],
        backup_files=[f"recovery/{manifest_id}/backup/accounting/rows.json"],
        content_hashes={},
        metadata={"originally_missing": []},
    )
    (tmp_path / "recovery" / f"{manifest_id}.json").write_text(
        manifest.model_dump_json(),
        encoding="utf-8",
    )

    with pytest.raises(LocalMasterRecoveryError) as excinfo:
        initialized_writer.recover(recovered_at=LATER, recovered_by=ActorSystem.LOCAL)

    assert isinstance(excinfo.value.__cause__, LocalMasterRecoveryError)
    assert read_manifest(tmp_path, manifest_id).phase == "RECOVERY_REQUIRED"
    assert kept.exists()
    with pytest.raises(LocalMasterRecoveryError):
        initialized_writer.recover(recovered_at=LATER, recovered_by=ActorSystem.LOCAL)


def test_rollback_failure_raises_recovery_error_and_keeps_all_evidence(
    initialized_writer: LocalMasterWriter,
    monkeypatch: pytest.MonkeyPatch,
    tmp_path: Path,
) -> None:
    lock = acquire(initialized_writer)

    def destroy_one_backup() -> None:
        for path in tmp_path.glob("recovery/*/backup/accounting/rows.json"):
            path.unlink()

    fail_replace_at(monkeypatch, "master_version.json", destroy_one_backup)
    with pytest.raises(LocalMasterRecoveryError) as excinfo:
        publish_accounting(initialized_writer, [accounting_row()], lock=lock)

    assert any(
        isinstance(error, OSError) and "simulated replace failure" in str(error)
        for error in exception_chain(excinfo.value)
    )
    assert excinfo.value.__cause__ is not None
    assert "RECOVERY_REQUIRED" in manifest_phases(tmp_path)
    remaining_backups = list(tmp_path.glob("recovery/*/backup/**/*"))
    assert any(path.name == "master_version.json" for path in remaining_backups)
    assert any((tmp_path / "staging").rglob("master_version.json"))

# Corruption: never repaired silently
# ---------------------------------------------------------------------------

CORRUPTION_CASES = [
    ("accounting/rows.json", "{broken", lambda w: w.read_accounting_rows()),
    ("banking/rows.json", "{broken", lambda w: w.read_banking_rows()),
    ("accounting/master_version.json", "{broken", lambda w: w.read_master_version(MasterType.ACCOUNTING)),
    ("banking/master_version.json", "{broken", lambda w: w.read_master_version(MasterType.BANKING)),
    ("audit/events.jsonl", "{broken\n", lambda w: w.read_audit_events()),
    ("audit/status_history.jsonl", "{broken\n", lambda w: w.read_status_history()),
    ("accounting/lock.json", "{broken", lambda w: w.read_lock(MasterType.ACCOUNTING)),
    ("accounting/rows.json", '[{"unexpected": 1}]', lambda w: w.read_accounting_rows()),
    ("banking/rows.json", '[{"unexpected": 1}]', lambda w: w.read_banking_rows()),
    ("accounting/master_version.json", '{"a": 1}', lambda w: w.read_master_version(MasterType.ACCOUNTING)),
    ("banking/master_version.json", '{"a": 1}', lambda w: w.read_master_version(MasterType.BANKING)),
    ("audit/events.jsonl", '{"a": 1}\n', lambda w: w.read_audit_events()),
    ("audit/status_history.jsonl", '{"a": 1}\n', lambda w: w.read_status_history()),
    ("accounting/lock.json", '{"a": 1}', lambda w: w.read_lock(MasterType.ACCOUNTING)),
]


@pytest.mark.parametrize(
    ("relative", "content", "reader"),
    CORRUPTION_CASES,
    ids=[f"{case[0]}-{index}" for index, case in enumerate(CORRUPTION_CASES)],
)
def test_corrupt_or_incompatible_files_raise_and_are_not_repaired(
    initialized_writer: LocalMasterWriter,
    tmp_path: Path,
    relative: str,
    content: str,
    reader: Any,
) -> None:
    target = tmp_path.joinpath(*relative.split("/"))
    target.write_text(content, encoding="utf-8")

    with pytest.raises(LocalMasterCorruptionError):
        reader(initialized_writer)

    assert target.read_text(encoding="utf-8") == content


def test_both_masters_roundtrip_exactly_with_decimals_dates_and_uuids(
    initialized_writer: LocalMasterWriter,
) -> None:
    accounting = mixed_row(
        company_contribution(),
        personal_contribution(),
        issue_date=date(2026, 10, 14),
    )
    batch = payment_batch(
        PaymentBatchType.GROUPED_TRANSFER,

            "amount": batch.total_amount,
            "unallocated_amount": batch.total_amount,

def write_pending_manifest(root: Path, phase: str) -> UUID:
    """Writes an unresolved manifest whose backup equals the current rows file."""
    manifest_id = uuid4()
    backup_relative = f"recovery/{manifest_id}/backup/accounting/rows.json"
    backup = root.joinpath(*backup_relative.split("/"))
    backup.parent.mkdir(parents=True)
    backup.write_bytes((root / "accounting" / "rows.json").read_bytes())
    manifest = LocalTransactionManifest(
        transaction_manifest_id=manifest_id,
        master_type=MasterType.ACCOUNTING,
        operation_id=uuid4(),
        idempotency_key=f"pending-{phase}",
        phase=phase,
        created_at=NOW,
        updated_at=NOW,
        target_files=["accounting/rows.json"],
        backup_files=[backup_relative],
        content_hashes={},
        metadata={"originally_missing": []},
    )
    (root / "recovery").mkdir(exist_ok=True)
    (root / "recovery" / f"{manifest_id}.json").write_text(
        manifest.model_dump_json(),
        encoding="utf-8",
    )
    return manifest_id


def crash_replace_once(
    monkeypatch: pytest.MonkeyPatch,
    matches: Any,
) -> None:
    original = writer_module.os.replace
    state = {"crashed": False}

    def hook(source: Any, target: Any) -> None:
        if not state["crashed"] and matches(Path(target)):
            state["crashed"] = True
            raise SimulatedCrash()
        original(source, target)

    monkeypatch.setattr(writer_module.os, "replace", hook)

# C. Interrupted initialize ----------------------------------------------------


@pytest.mark.parametrize("phase", ["PREPARED", "REPLACING"])
def test_interrupted_initialize_is_recovered_to_the_uninitialized_state(
    tmp_path: Path,
    monkeypatch: pytest.MonkeyPatch,
    phase: str,
) -> None:
    writer = LocalMasterWriter(tmp_path)
    manifest_writes = {"count": 0}

    def second_manifest_write(target: Path) -> bool:
        if target.parent.name == "recovery" and target.suffix == ".json":
            manifest_writes["count"] += 1
            return manifest_writes["count"] == 2
        return False

    def accounting_version_replace(target: Path) -> bool:
        return target.parent.name == "accounting" and target.name == "master_version.json"

    crash_replace_once(
        monkeypatch,
        second_manifest_write if phase == "PREPARED" else accounting_version_replace,
    )
    with pytest.raises(SimulatedCrash):
        initialize_writer(writer)
    monkeypatch.undo()

    assert manifest_phases(tmp_path) == [phase]
    assert (tmp_path / "accounting" / "rows.json").exists() is (phase == "REPLACING")
    with pytest.raises(LocalMasterInitializationError):
        initialize_writer(writer)

    recovered = writer.recover(recovered_at=LATER, recovered_by=ActorSystem.LOCAL)

    assert len(recovered) == 1
    assert manifest_phases(tmp_path) == ["ROLLED_BACK"]
    assert [name for name in REQUIRED_STATE_FILES if (tmp_path / name).exists()] == []
    initialize_writer(writer)
    assert writer.read_master_version(MasterType.ACCOUNTING).current_version == 0
    assert writer.read_master_version(MasterType.BANKING).current_version == 0
    assert writer.recover(recovered_at=LATER, recovered_by=ActorSystem.LOCAL) == []


# D. Operation timestamps before the lock timestamps ---------------------------


@pytest.mark.parametrize("action", ["release", "publish"])
@pytest.mark.parametrize("reference", ["acquired_at", "heartbeat_at"])
def test_operation_before_lock_timestamps_is_refused_before_writing(
    initialized_writer: LocalMasterWriter,
    tmp_path: Path,
    action: str,
    reference: str,
) -> None:
    held = acquire(initialized_writer)
    if reference == "heartbeat_at":
        renewed_at = held.acquired_at + timedelta(minutes=10)
        initialized_writer.renew_lock(
            MasterType.ACCOUNTING,
            lock_id=held.lock_id,
            owner_system=held.owner_system,

            now=renewed_at,
            new_expires_at=renewed_at + timedelta(hours=2),
        )
        operation_time = held.acquired_at + timedelta(minutes=5)
    else:
        operation_time = held.acquired_at - timedelta(minutes=5)
    before = snapshot(tmp_path)

    with pytest.raises(MasterTableError):
        if action == "release":
            initialized_writer.release_lock(
                MasterType.ACCOUNTING,
                lock_id=held.lock_id,
                owner_system=held.owner_system,
                owner_instance_id=held.owner_instance_id,
                released_at=operation_time,
            )
        else:
            publish_accounting(
                initialized_writer,
                [accounting_row()],
                lock=held,
                committed_at=operation_time,
            )

    assert snapshot(tmp_path) == before
    current = initialized_writer.read_lock(MasterType.ACCOUNTING)
    assert current is not None
    assert current.lock_id == held.lock_id
    assert current.status == LockStatus.ACTIVE

_CURRENCY_PATTERN = re.compile(r"[A-Z]{3}", re.ASCII)
_IBAN_PATTERN = re.compile(r"[A-Z0-9]{15,34}", re.ASCII)
_SIRET_PATTERN = re.compile(r"[0-9]{14}", re.ASCII)
_HASH_PATTERN = re.compile(r"[0-9a-fA-F]{64}", re.ASCII)
_MONEY_TOLERANCE = Decimal("0.01")


def _decimal_from_value(value: Any) -> Decimal:
    if isinstance(value, Decimal):
        return value
    try:
        return Decimal(str(value))
    except (InvalidOperation, ValueError, TypeError) as exc:
        raise ValueError("value must be a valid decimal") from exc


def _validate_currency(value: str) -> str:
    if not _CURRENCY_PATTERN.fullmatch(value):
        raise ValueError("currency must contain exactly three uppercase ASCII letters")
    return value


def _normalize_iban(value: str | None) -> str | None:
    if value is None:
        return None
    normalized = "".join(value.split()).upper()
    if not _IBAN_PATTERN.fullmatch(normalized):
        raise ValueError("IBAN must contain 15 to 34 ASCII alphanumeric characters")
    return normalized


def _require_nonblank(value: str | None, field_name: str) -> str | None:
    if value is not None and not value.strip():
        raise ValueError(f"{field_name} must not be blank")
    return value

class PaymentBatchItem(_StrictBankingModel):
    item_id: UUID
    counterparty: BankCounterparty | None = None
    observed_name: str | None = None
    effect_reference: str | None = None
    invoice_reference: str | None = None
    due_date: date | None = None
    amount: Decimal
    status: PaymentItemStatus
    allocations: list[ReconciliationAllocation] = Field(default_factory=list)
    unallocated_amount: Decimal
    confidence_score: float | None = None
    source_evidence_ref: str | None = None
    metadata: dict[str, Any] = Field(default_factory=dict)

    @field_validator("amount", "unallocated_amount", mode="before")
    @classmethod
    def convert_money(cls, value: Any) -> Decimal:
        return _decimal_from_value(value)

    @field_validator("amount")
    @classmethod
    def validate_amount(cls, value: Decimal) -> Decimal:
        if not value.is_finite() or value <= 0:
            raise ValueError("amount must be positive and finite")
        return value

    @field_validator("unallocated_amount")
    @classmethod
    def validate_unallocated_amount(cls, value: Decimal) -> Decimal:
        if not value.is_finite() or value < 0:
            raise ValueError("unallocated_amount must be non-negative and finite")
        return value

    @field_validator(
        "observed_name",
        "effect_reference",
        "invoice_reference",
        "source_evidence_ref",
    )
    @classmethod
    def validate_optional_text(cls, value: str | None) -> str | None:
        return _require_nonblank(value, "text")

    @field_validator("confidence_score")
    @classmethod
    def validate_confidence_score(cls, value: float | None) -> float | None:
        if value is not None and (not 0 <= value <= 1):
            raise ValueError("confidence_score must be between 0 and 1")
        return value

    @model_validator(mode="after")
    def validate_item_consistency(self) -> PaymentBatchItem:
        allocated_amount = sum(
            (allocation.allocated_amount for allocation in self.allocations),
            start=Decimal("0"),
        )
        if allocated_amount > self.amount:
            raise ValueError("allocations cannot exceed item amount")
        expected_unallocated = self.amount - allocated_amount
        if abs(self.unallocated_amount - expected_unallocated) > _MONEY_TOLERANCE:
            raise ValueError("unallocated_amount does not match item allocations")

        if self.status == PaymentItemStatus.MATCHED:
            if self.unallocated_amount != 0 or allocated_amount != self.amount:
                raise ValueError("MATCHED item requires allocations covering its amount")
            if any(
                allocation.allocation_status != ReconciliationStatus.MATCHED
                for allocation in self.allocations
            ):
                raise ValueError("MATCHED item requires validated allocations")
        elif self.status == PaymentItemStatus.PARTIALLY_MATCHED:
            if self.unallocated_amount <= 0 or allocated_amount <= 0:
                raise ValueError("PARTIALLY_MATCHED item requires allocations and a remainder")
        return self


class PaymentBatch(_StrictBankingModel):
    batch_id: UUID
    batch_type: PaymentBatchType
    batch_reference: str | None = None
    booking_date: date
    total_amount: Decimal
    currency: str
    expected_item_count: int | None = None
    items: list[PaymentBatchItem] = Field(default_factory=list)
    status: PaymentBatchStatus
    unexplained_amount: Decimal
    source_evidence_refs: list[str] = Field(default_factory=list)
    created_at: datetime
    updated_at: datetime
    metadata: dict[str, Any] = Field(default_factory=dict)

    @field_validator("total_amount", "unexplained_amount", mode="before")
    @classmethod
    def convert_money(cls, value: Any) -> Decimal:
        return _decimal_from_value(value)

    @field_validator("total_amount")
    @classmethod
    def validate_total_amount(cls, value: Decimal) -> Decimal:
        if not value.is_finite() or value <= 0:
            raise ValueError("total_amount must be positive and finite")
        return value

    @field_validator("unexplained_amount")
    @classmethod
    def validate_unexplained_amount(cls, value: Decimal) -> Decimal:
        if not value.is_finite() or value < 0:
            raise ValueError("unexplained_amount must be non-negative and finite")
        return value

    @field_validator("currency")
    @classmethod
    def validate_currency(cls, value: str) -> str:
        return _validate_currency(value)

    @field_validator("batch_reference")
    @classmethod
    def validate_batch_reference(cls, value: str | None) -> str | None:
        return _require_nonblank(value, "batch_reference")

    @field_validator("expected_item_count")
    @classmethod
    def validate_expected_item_count(cls, value: int | None) -> int | None:
        if value is not None and value <= 0:
            raise ValueError("expected_item_count must be positive")
        return value

    @field_validator("source_evidence_refs")
    @classmethod
    def validate_source_evidence_refs(cls, values: list[str]) -> list[str]:
        if any(not value.strip() for value in values):
            raise ValueError("source_evidence_refs must not contain blank values")
        return values

    @field_validator("created_at", "updated_at")
    @classmethod
    def validate_timestamps(cls, value: datetime) -> datetime:
        if value.tzinfo is None or value.utcoffset() != timedelta(0):
            raise ValueError("timestamps must be timezone-aware UTC")
        return value

    @model_validator(mode="after")
    def validate_batch_consistency(self) -> PaymentBatch:
        try:
            timestamps_are_out_of_order = self.updated_at < self.created_at
        except TypeError as exc:
            raise ValueError("created_at and updated_at must be comparable") from exc
        if timestamps_are_out_of_order:
            raise ValueError("updated_at cannot precede created_at")

        items_total = sum(
            (item.amount for item in self.items),
            start=Decimal("0"),
        )
        if items_total > self.total_amount:
            raise ValueError("items cannot exceed batch total_amount")
        expected_unexplained = self.total_amount - items_total
        if abs(self.unexplained_amount - expected_unexplained) > _MONEY_TOLERANCE:
            raise ValueError("unexplained_amount does not match item total")

        if self.expected_item_count is not None and len(self.items) > self.expected_item_count:
            raise ValueError("items cannot exceed expected_item_count")

        if self.status == PaymentBatchStatus.DETAIL_MISSING and self.items:
            raise ValueError("DETAIL_MISSING batch cannot contain items")
        if self.status == PaymentBatchStatus.PARTIALLY_DETAILED:
            if not self.items or self.unexplained_amount <= 0:
                raise ValueError("PARTIALLY_DETAILED requires items and a positive remainder")
        if self.status == PaymentBatchStatus.RECONCILED:
            if items_total != self.total_amount or self.unexplained_amount != 0:
                raise ValueError("RECONCILED requires items covering the full batch total")
            if any(item.status != PaymentItemStatus.MATCHED for item in self.items):
                raise ValueError("RECONCILED requires all items to be MATCHED")
        return self

    allocations: list[ReconciliationAllocation] = Field(default_factory=list)
    payment_batch: PaymentBatch | None = None
    unallocated_amount: Decimal
    statement_id: UUID | None = None
    source_file: str | None = None
    import_batch_id: UUID
    source_evidence_ids: list[UUID] = Field(default_factory=list)
    row_version: StrictInt
    content_hash: str
    is_deleted: bool = False
    deleted_at: datetime | None = None
    deleted_by: str | None = None
    deletion_reason: str | None = None
    created_at: datetime
    updated_at: datetime
    metadata: dict[str, Any] = Field(default_factory=dict)

    @field_validator("amount", "unallocated_amount", mode="before")
    @classmethod
    def convert_money(cls, value: Any) -> Decimal:
        return _decimal_from_value(value)

    @field_validator("amount")
    @classmethod
    def validate_amount(cls, value: Decimal) -> Decimal:
        if not value.is_finite() or value <= 0:
            raise ValueError("amount must be positive and finite")
        return value

    @field_validator("unallocated_amount")
    @classmethod
    def validate_unallocated_amount(cls, value: Decimal) -> Decimal:
        if not value.is_finite() or value < 0:
            raise ValueError("unallocated_amount must be non-negative and finite")
        return value

    @field_validator("currency")
    @classmethod
    def validate_currency(cls, value: str) -> str:
        return _validate_currency(value)

    @field_validator("raw_label")
    @classmethod
    def validate_raw_label(cls, value: str) -> str:
        if not value.strip():
            raise ValueError("raw_label must not be blank")
        return value

    @field_validator("row_version")
    @classmethod
    def validate_row_version(cls, value: int) -> int:
        if value <= 0:
            raise ValueError("row_version must be positive")
        return value

    @field_validator("content_hash")
    @classmethod
    def validate_content_hash(cls, value: str) -> str:
        if not _HASH_PATTERN.fullmatch(value):
            raise ValueError("content_hash must be a 64-character SHA-256 hex digest")
        return value

    @field_validator("normalized_label", "bank_reference", "source_file")
    @classmethod
    def validate_optional_text(cls, value: str | None) -> str | None:
        return _require_nonblank(value, "text")

    @field_validator("deleted_by", "deletion_reason")
    @classmethod
    def validate_deletion_text(cls, value: str | None) -> str | None:
        return _require_nonblank(value, "deletion detail")

    @field_validator("created_at", "updated_at", "deleted_at")
    @classmethod
    def validate_utc_timestamps(cls, value: datetime | None) -> datetime | None:
        if value is not None and (
            value.tzinfo is None or value.utcoffset() != timedelta(0)
        ):
            raise ValueError("timestamps must be timezone-aware UTC")
        return value

    @model_validator(mode="after")
    def validate_record_consistency(self) -> "BankingRecord":
        try:
            timestamp_is_out_of_order = self.updated_at < self.created_at
        except TypeError as exc:
            raise ValueError("created_at and updated_at must be comparable") from exc
        if timestamp_is_out_of_order:
            raise ValueError("updated_at cannot precede created_at")
        if self.is_deleted:
            if (
                self.deleted_at is None
                or self.deleted_by is None
                or self.deletion_reason is None
            ):
                raise ValueError(
                    "soft deletion requires deleted_at, deleted_by, and deletion_reason"
                )
        elif any(
            value is not None
            for value in (self.deleted_at, self.deleted_by, self.deletion_reason)
        ):
            raise ValueError("deleted_* fields must be absent when is_deleted is false")

        if (
            self.cheque is not None
            and self.transaction_type
            not in {BankTransactionType.CHEQUE, BankTransactionType.CHEQUE_DEPOSIT}
        ):
            raise ValueError("cheque is only valid for cheque transaction types")

        if self.payment_batch is not None:
            if self.allocations:
                raise ValueError(
                    "direct allocations and payment_batch cannot be combined"
                )
            if abs(self.payment_batch.total_amount - self.amount) > _MONEY_TOLERANCE:
                raise ValueError("payment_batch total must match transaction amount")
            if self.transaction_type == BankTransactionType.CHEQUE_DEPOSIT:
                if self.payment_batch.batch_type != PaymentBatchType.CHEQUE_DEPOSIT:
                    raise ValueError(
                        "CHEQUE_DEPOSIT transaction requires a CHEQUE_DEPOSIT batch"
                    )

        batch_allocations = (
            [
                allocation
                for item in self.payment_batch.items
                for allocation in item.allocations
            ]
            if self.payment_batch is not None
            else []
        )
        # With a payment batch the item allocations are the only source of truth;
        # direct allocations are forbidden above, so this never double counts.
        effective_allocations = [*self.allocations, *batch_allocations]
        allocated_amount = sum(
            (allocation.allocated_amount for allocation in effective_allocations),
            start=Decimal("0"),
        )
        if allocated_amount > self.amount:
            raise ValueError("allocations cannot exceed amount")

        expected_unallocated = self.amount - allocated_amount
        if abs(self.unallocated_amount - expected_unallocated) > _MONEY_TOLERANCE:
            raise ValueError("unallocated_amount does not match the allocation total")

        if not effective_allocations:
            allowed_without_allocations = {
                ReconciliationStatus.UNMATCHED,
                ReconciliationStatus.PROPOSED,
                ReconciliationStatus.NEEDS_REVIEW,
                ReconciliationStatus.REJECTED,
            }
            if self.reconciliation_status not in allowed_without_allocations:
                raise ValueError("reconciliation status requires allocations")

        if allocated_amount > 0 and allocated_amount < self.amount:
            if self.reconciliation_status not in {
                ReconciliationStatus.PARTIALLY_MATCHED,
                ReconciliationStatus.NEEDS_REVIEW,
            }:
                raise ValueError(
                    "partial allocations require PARTIALLY_MATCHED or NEEDS_REVIEW"
                )

        if self.reconciliation_status == ReconciliationStatus.MATCHED:
            if self.unallocated_amount != 0 or allocated_amount != self.amount:
                raise ValueError("MATCHED requires allocations covering the full amount")
            if any(
                allocation.allocation_status != ReconciliationStatus.MATCHED
                for allocation in effective_allocations
            ):
                raise ValueError("MATCHED requires validated allocations")

        if self.payment_batch is not None:
            batch_reconciled = self.payment_batch.status == PaymentBatchStatus.RECONCILED
            parent_matched = self.reconciliation_status == ReconciliationStatus.MATCHED
            if parent_matched and not batch_reconciled:
                raise ValueError("MATCHED parent requires a RECONCILED payment batch")
            if batch_reconciled and not parent_matched:
                raise ValueError("RECONCILED payment batch requires a MATCHED parent")

        if (
            self.reconciliation_status == ReconciliationStatus.PARTIALLY_MATCHED
            and self.unallocated_amount <= 0
        ):
            raise ValueError("PARTIALLY_MATCHED requires a positive unallocated amount")

        return self

BATCH_SCENARIOS = {
    "lcr": (
        PaymentBatchType.LCR_REMITTANCE,
        BankTransactionType.TRANSFER,
        PaymentMethod.TRANSFER,
        ["4250.00"] * 10,
    ),
    "cheque-deposit": (
        PaymentBatchType.CHEQUE_DEPOSIT,
        BankTransactionType.CHEQUE_DEPOSIT,
        PaymentMethod.CHEQUE,
        ["10000.00", "12500.00", "20000.00"],
    ),
}


def scenario_items(amounts: list[str], matched_count: int) -> list[PaymentBatchItem]:
    items: list[PaymentBatchItem] = []
    for index, amount in enumerate(amounts):
        matched = index < matched_count
        items.append(
            batch_item(
                item_id=UUID(int=500 + index),
                amount=amount,
                status=(
                    PaymentItemStatus.MATCHED if matched else PaymentItemStatus.UNMATCHED
                ),
                allocations=(
                    [
                        allocation(
                            allocation_id=UUID(int=600 + index),
                            target_id=UUID(int=700 + index),
                            allocated_amount=amount,
                        )
                    ]
                    if matched
                    else []
                ),
                unallocated_amount="0" if matched else amount,
            )
        )
    return items


def scenario_record(
    scenario: str,
    *,
    matched_count: int,
    batch_status: PaymentBatchStatus,
    reconciliation_status: ReconciliationStatus,
    unallocated_amount: str,
) -> BankingRecord:
    batch_type, transaction_type, method, amounts = BATCH_SCENARIOS[scenario]
    total = sum((Decimal(item) for item in amounts), Decimal("0"))
    overrides: dict[str, Any] = {}
    if transaction_type == BankTransactionType.CHEQUE_DEPOSIT:
        overrides["cheque"] = ChequeReference(
            deposit_reference="REM-1",
            cheque_count=len(amounts),
            total_deposit_amount=total,
        )
    return record(
        direction=BankTransactionDirection.CREDIT,
        transaction_type=transaction_type,
        payment_method=method,
        amount=total,
        payment_batch=payment_batch(
            batch_type=batch_type,
            total_amount=total,
            expected_item_count=len(amounts),
            items=scenario_items(amounts, matched_count),
            status=batch_status,
            unexplained_amount="0",
        ),
        reconciliation_status=reconciliation_status,
        unallocated_amount=unallocated_amount,
        **overrides,
    )
