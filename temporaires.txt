# DSH-Core — document unique de reprise et état exécutif

## 1. Objectif final DSH-Core
Construire un pipeline local de traitement documentaire qui oriente les pages, effectue l'OCR, extrait des données structurées, puis applique validation canonique et contrôles métier.

## 2. Architecture cible complète
- Rendu du document → orientation de page → OCR → extraction structurée → normalisation/validation canonique → contrôles métier.
- PP-LCNet_x1_0_doc_ori est le candidat local provisoire d'orientation ; l'OCR suit l'orientation.
- ExtractionManager/Ollama est le chemin d'extraction intégré à évaluer ; NuExtract3 est un candidat d'extraction structuré après OCR, pas un moteur OCR.
- Comparer les extracteurs sur les mêmes textes OCR et la vérité terrain des champs fournisseur, date, HT, TVA, TTC et SIRET.
- Exclure réconciliation, règles TVA et skills du benchmark d'exactitude brute ; les évaluer séparément.
- Les modèles cloud sont des secours explicites pour les cas complexes, jamais le chemin par défaut des tâches locales simples.

## 3. État exact des environnements
- **DSH-Core :** dépôt applicatif et pipeline cible. Les documents consultés ne consignent ni environnement d'exécution, ni versions de runtime/dépendances ; ne pas les supposer.
- **DeepSeek Harness :** DeepSeek Harness est validé et conservé comme composant stratégique du projet. Les documents consultés ne précisent ni chemin, ni versions, ni périmètre/critères de validation ; ne pas installer ou revalider sans besoin précis.
- **Paddle Orientation :** environnement isolé `A:\dsh-paddle-orientation` ; Python 3.11.9, PaddlePaddle GPU 3.0.0, PaddleOCR 3.0.3, PaddleX 3.0.3, CUDA 12.6, RTX 4070 Laptop. Hors dépôt DSH-Core.
- **LightOnOCR Bench :** environnement isolé `A:\dsh-lightonocr-bench` ; Python 3.11.9, Torch 2.14.1+cu126, cache local du modèle d'environ 2,33 Go. Hors dépôt DSH-Core.

===== model_router.py =====
"""Central model selection without loading dotenv files or contacting providers."""

import os
from dataclasses import dataclass
from enum import Enum
from typing import Mapping, Optional, Union


class ModelRole(str, Enum):
    FAST = "fast"
    BASE = "base"
    VISION = "vision"
    EXTRACTION = "extraction"
    EMBEDDING = "embedding"
    PREPARATION = "preparation"
    EXPERT = "expert"
    CODING = "coding"
    REVIEW = "review"


class ExecutionMode(str, Enum):
    LOCAL = "local"
    HYBRID = "hybrid"
    CLOUD = "cloud"


class Provider(str, Enum):
    OLLAMA = "ollama"
    OPENROUTER = "openrouter"


@dataclass(frozen=True)
class ConfigIssue:
    variable: str
    role: Optional[ModelRole]
    mode: Optional[ExecutionMode]
    message: str


RoleInput = Union[ModelRole, str]
ModeInput = Union[ExecutionMode, str]

_ROLE_VARIABLES = {
    ExecutionMode.LOCAL: {
        ModelRole.FAST: "LOCAL_FAST_MODEL",
        ModelRole.BASE: "LOCAL_BASE_MODEL",
        ModelRole.VISION: "LOCAL_VISION_MODEL",
        ModelRole.EXTRACTION: "LOCAL_EXTRACTION_MODEL",
        ModelRole.EMBEDDING: "LOCAL_EMBEDDING_MODEL",
        ModelRole.PREPARATION: "LOCAL_BASE_MODEL",
        ModelRole.EXPERT: "LOCAL_BASE_MODEL",
        ModelRole.CODING: "LOCAL_BASE_MODEL",
        ModelRole.REVIEW: "LOCAL_BASE_MODEL",
    },
    ExecutionMode.HYBRID: {
        ModelRole.FAST: "LOCAL_FAST_MODEL",
        ModelRole.BASE: "LOCAL_BASE_MODEL",
        ModelRole.VISION: "LOCAL_VISION_MODEL",
        ModelRole.EXTRACTION: "LOCAL_EXTRACTION_MODEL",
        ModelRole.EMBEDDING: "LOCAL_EMBEDDING_MODEL",
        ModelRole.PREPARATION: "CLOUD_PREPARATION_MODEL",
        ModelRole.EXPERT: "CLOUD_EXPERT_MODEL",
        ModelRole.CODING: "CLOUD_CODING_MODEL",
        ModelRole.REVIEW: "CLOUD_REVIEW_MODEL",
    },
    ExecutionMode.CLOUD: {
        ModelRole.VISION: "CLOUD_VISION_MODEL",
        ModelRole.PREPARATION: "CLOUD_PREPARATION_MODEL",
        ModelRole.EXPERT: "CLOUD_EXPERT_MODEL",
        ModelRole.CODING: "CLOUD_CODING_MODEL",
        ModelRole.REVIEW: "CLOUD_REVIEW_MODEL",
    },
}

_CLOUD_VARIABLES = frozenset(
    variable
    for variables in _ROLE_VARIABLES.values()
    for variable in variables.values()
    if variable.startswith("CLOUD_")
)
_RETIRED_MODELS = frozenset({"qwen2.5vl:7b", "qwen3.5:2b"})


def validate_model_identifier(model: str) -> str:
    """Reject retired model identifiers without loading a model."""
    if not model or not model.strip():
        raise ValueError("MODEL_NOT_CONFIGURED")
    if model.strip().casefold() in _RETIRED_MODELS:
        raise ValueError(f"MODEL_RETIRED: {model.strip()}")
    return model.strip()


def validate_vision_model(model: str) -> str:
    """Reject missing or retired Vision selections without loading a model."""
    try:
        return validate_model_identifier(model)
    except ValueError as exc:
        if str(exc).startswith("MODEL_RETIRED") or str(exc) == "MODEL_NOT_CONFIGURED":
            raise ValueError("VISION_MODEL_NOT_CONFIGURED") from None
        raise


def _parse_role(role: RoleInput) -> ModelRole:
    try:
        return role if isinstance(role, ModelRole) else ModelRole(role)
    except (TypeError, ValueError):
        raise ValueError(f"Unknown model role: {role!r}") from None


def _parse_mode(mode: ModeInput) -> ExecutionMode:
    try:
        return mode if isinstance(mode, ExecutionMode) else ExecutionMode(mode)
    except (TypeError, ValueError):
        raise ValueError(f"Unknown execution mode: {mode!r}") from None


def _environment(environ: Optional[Mapping[str, str]]) -> Mapping[str, str]:
    return os.environ if environ is None else environ


def _selected_mode(
    mode: Optional[ModeInput], environ: Mapping[str, str]
) -> ExecutionMode:
    if mode is not None:
        return _parse_mode(mode)
    configured_mode = environ.get("AI_MODE")
    if configured_mode is None or not configured_mode.strip():
        raise ValueError("AI_MODE is missing or empty")
    return _parse_mode(configured_mode)


def _variable_for(role: ModelRole, mode: ExecutionMode) -> str:
    variable = _ROLE_VARIABLES[mode].get(role)
    if variable is None:
        raise ValueError(f"Role {role.value!r} is not supported in mode {mode.value!r}")
    return variable


def resolve_model(
    role: RoleInput,
    mode: Optional[ModeInput] = None,
    environ: Optional[Mapping[str, str]] = None,
) -> str:
    """Return a configured model identifier, never a credential."""
    role_value = _parse_role(role)
    env = _environment(environ)
    mode_value = _selected_mode(mode, env)
    variable = _variable_for(role_value, mode_value)
    model = env.get(variable)
    if model is None or not model.strip():
        if role_value is ModelRole.VISION:
            raise ValueError("VISION_MODEL_NOT_CONFIGURED")
        raise ValueError(
            f"Missing or empty {variable} for role {role_value.value!r} "
            f"in mode {mode_value.value!r}"
        )
    if role_value is ModelRole.VISION:
        return validate_vision_model(model)
    return validate_model_identifier(model)


def resolve_provider(role: RoleInput, mode: Optional[ModeInput] = None) -> Provider:
    """Return the provider assigned by the role/mode table."""
    role_value = _parse_role(role)
    mode_value = _selected_mode(mode, os.environ)
    variable = _variable_for(role_value, mode_value)
    return Provider.OPENROUTER if variable.startswith("CLOUD_") else Provider.OLLAMA


def validate_model_config(
    mode: Optional[ModeInput] = None,
    environ: Optional[Mapping[str, str]] = None,
) -> list[ConfigIssue]:
    """Report missing configuration without exposing environment values."""
    env = _environment(environ)
    if mode is None:
        configured_mode = env.get("AI_MODE")
        if configured_mode is None or not configured_mode.strip():
            return [
                ConfigIssue(
                    variable="AI_MODE",
                    role=None,
                    mode=None,
                    message="AI_MODE is missing or empty",
                )
            ]
        try:
            mode_value = _parse_mode(configured_mode)
        except ValueError:
            return [
                ConfigIssue(
                    variable="AI_MODE",
                    role=None,
                    mode=None,
                    message="AI_MODE contains an unknown execution mode",
                )
            ]
    else:
        mode_value = _parse_mode(mode)

    issues = []
    role_variables = _ROLE_VARIABLES[mode_value]
    checked_variables = set()
    for role_value, variable in role_variables.items():
        if variable not in checked_variables and (
            variable not in env or not env[variable].strip()
        ):
            issues.append(
                ConfigIssue(
                    variable=variable,
                    role=role_value,
                    mode=mode_value,
                    message=f"{variable} is missing or empty",
                )
            )
        if variable is not None:
            checked_variables.add(variable)

    if any(variable.startswith("CLOUD_") for variable in role_variables.values()):
        if not env.get("OPENROUTER_API_KEY", "").strip():
            issues.append(
                ConfigIssue(
                    variable="OPENROUTER_API_KEY",
                    role=None,
                    mode=mode_value,
                    message="OPENROUTER_API_KEY is missing or empty",
                )
            )
    return issues

import re
from typing import Any, Dict
from model_router import ExecutionMode, ModelRole, resolve_model

# 1. REGISTRE DES MODÈLES (FACILE À MODIFIER)
# ==========================================
MODELS_REGISTRY = {
    "vision_general": "",                  # Résolu par model_router ; aucun fallback
    "executor_fast": "qwen3.5:9b",          # Tâches courantes, scripts rapides, MCP
    "reasoning_expert": "qwen3.6:27b",      # Raisonnement lourd, refactoring complexe
    "supervisor_cloud": "anthropic/claude-3-5-sonnet" # Audit, architecture, validation de SKILL
}

# Import sécurisé du gestionnaire unifié
try:
    from memory_manager import recuperer_contexte, find_skill
    HAS_MEMORY_MANAGER = True
except ImportError:
    HAS_MEMORY_MANAGER = False

def detecter_complexite(prompt: str) -> str:
    """Évalue si la tâche locale nécessite l'expert Qwen 27B ou l'exécuteur 9B."""
    prompt_lower = prompt.lower()
    
    expert_patterns = [
        r'\b(optimis|refactor|complex|algorithme|debug.*avancé)\b',
        r'\b(architecture.*données|parcours.*graphe|concurren)\b',
        r'\b(analyse.*critique|modélis|pipeline.*comptable)\b'
    ]
    if any(re.search(p, prompt_lower) for p in expert_patterns):
        return "expert"
    return "fast"

def analyze_prompt_complexity(prompt: str, image_provided: bool = False) -> Dict[str, str]:
    """Détermine le niveau hiérarchique requis selon la nature du prompt."""
    prompt_lower = prompt.lower()

    # 1. Flux Multimodal / Vision (Aiguillage général)
    vision_patterns = [r'\b(image|photo|capture|scann|vision|vois-tu|décris)\b']
    if image_provided or any(re.search(p, prompt_lower) for p in vision_patterns):
        return {
            "agent": "vision-sensor-local",
            "model": resolve_model(ModelRole.VISION, ExecutionMode.LOCAL),
            "role": "Capteur Visuel (Multimodal)",
            "tier": "vision",
            "reasoning": "Présence d'une image ou requête d'observation visuelle générale."
        }

    # 2. Superviseur Cloud (Audit, gouvernance, rédaction de règles / SKILL)
    teacher_patterns = [
        r'\b(skill|compétence|mémoire procédurale|template)\b',
        r'\b(juge|audit.*sécurité|valide.*code|contrôle.*qualité)\b',
        r'\b(arbitre|master.*blueprint|décision.*architecture)\b'
    ]
    if any(re.search(p, prompt_lower) for p in teacher_patterns):
        return {
            "agent": "cloud-supervisor",
            "model": MODELS_REGISTRY["supervisor_cloud"],
            "role": "Superviseur & Rédacteur SKILL",
            "tier": "supervisor",
            "reasoning": "Tâche d'arbitrage de haut niveau, audit de code ou gouvernance du système."
        }

    # 3. Cascade Locale (Expert 27B vs Exécuteur 9B)
    if detecter_complexite(prompt) == "expert":
        return {
            "agent": "qwen-expert-local",
            "model": MODELS_REGISTRY["reasoning_expert"],
            "role": "Expert Raisonnement Local",
            "tier": "expert",
            "reasoning": "Tâche locale nécessitant une capacité d'abstraction et d'analyse avancée."
        }

    return {
        "agent": "qwen-executor-local",
        "model": MODELS_REGISTRY["executor_fast"],
        "role": "Exécuteur Rapide Local",
        "tier": "fast",
        "reasoning": "Tâche opérationnelle directe, manipulation d'outils MCP ou script rapide."
    }

def route_prompt(prompt: str, user_id: str = "admin", image_provided: bool = False) -> Dict[str, Any]:
    """Aiguille la requête et injecte les souvenirs et compétences applicables."""
    decision = analyze_prompt_complexity(prompt, image_provided)
    
    # 1. Vérification d'une compétence outillée (Skill)
    matched_skill = None
    if HAS_MEMORY_MANAGER:
        matched_skill = find_skill(prompt)

    # 2. Récupération du contexte contextuel utilisateur
    contexte_utilisateur = ""
    if HAS_MEMORY_MANAGER:
        contexte_utilisateur = recuperer_contexte(user_id, prompt)

    # 3. Assemblage du contrat d'instructions
    instructions = {
        "vision": "Tu es le Capteur Visuel. Analyse les éléments graphiques, extrait le texte visible et décris la scène avec précision.",
        "supervisor": "Tu es le Superviseur. Audite les résultats, valide le respect des normes et génère les fiches SKILL.",
        "expert": "Tu es l'Expert Local. Résous les blocages logiques complexes, optimise le code et garantis la cohérence technique.",
        "fast": "Tu es l'Exécuteur Rapide. Utilise les outils locaux (MCP, fichiers, scripts) pour exécuter l'action demandée sans détour."
    }

    return {
        **decision,
        "instructions": instructions[decision["tier"]],
        "active_skill": matched_skill,
        "episodic_context": contexte_utilisateur,
        "input_prompt": prompt
    }

if __name__ == "__main__":

class InvoiceExtractionValidationError(Exception):

class ExtractionManager:
    def __init__(self, base_model: str = None):
        self.base_model = (
            validate_model_identifier(base_model)
            if base_model is not None

    def extract_to_json(self, document: DocumentResult) -> dict:
        print(f"\n[ExtractionManager] Structuration du texte brut via {self.base_model}...")

        # Le prompt est mis à jour pour coller EXACTEMENT à ton InvoiceSchema
        prompt = f"""Tu es un expert comptable implacable. Transforme ce texte brut extrait d'un document en un JSON strict qui respecte EXACTEMENT cette structure.
        
Origine du texte : {document.profile_type} extrait via {document.extraction_tool}
Texte brut :
{document.raw_text}

Format JSON attendu (réponds UNIQUEMENT le JSON, aucun texte avant ou après) :
{{
  "numero_facture": "Numéro ou FA-INCONNU",
  "date_emission": "YYYY-MM-DD",

    "nom": "Nom du vendeur"
  }},

  "lignes": [
    {{
      "description": "Nom de l'article",
      "quantite": 1.0,
      "prix_unitaire_ht": 0.0,
      "total_ht": 0.0,
      "taux_tva": 20.0
    }}
  ],
  "total_ht": 0.0,
  "montant_tva": 0.0,
  "total_ttc": 0.0,
  "devise": "EUR"
}}"""

        try:
            response = ollama.chat(
                model=self.base_model,
                messages=[{"role": "user", "content": prompt}],
                options={"temperature": 0.0}
            )
            
            raw_json = response["message"]["content"].strip()
            
            # Nettoyage si le modèle ajoute des balises Markdown
            if raw_json.startswith("```json"):
                raw_json = raw_json.replace("```json", "").replace("```", "").strip()
            elif raw_json.startswith("```"):
                raw_json = raw_json.replace("```", "").strip()

            parsed_data = json.loads(raw_json)
            return self._validate_data(parsed_data)

        except json.JSONDecodeError:
            print("[ExtractionManager] ❌ ÉCHEC : Le modèle n'a pas renvoyé un JSON lisible.")
            return {"status": "error", "message": "JSON invalide", "raw_output": raw_json}
        except Exception as e:

            return {"status": "critical_error", "message": str(e)}

            error_msgs = [err["msg"] for err in validation_error.errors()]
            return {"status": "validation_failed", "extracted_data": data, "errors": error_msgs}

# services/routing_engine.py
import pandas as pd
import pymupdf
import ollama
from pathlib import Path
from model_router import ExecutionMode, ModelRole, resolve_model, validate_vision_model
from schemas.profile_schema import DocumentProfile
from schemas.document_result import DocumentResult

class RoutingEngine:
    def __init__(self, vision_model: str | None = None):
        self.vision_model = (
            validate_vision_model(vision_model) if vision_model is not None else None
        )

    def extract(self, file_path: str | Path, profile: DocumentProfile) -> DocumentResult:
        """Route le document vers le bon outil et retourne l'enveloppe DocumentResult."""
        path = Path(file_path)
        ptype = profile.profile_type
        tool_used = "unknown"
        raw_text = ""
        notes = []

        print(f"\n[RoutingEngine] Routage de {path.name} -> {ptype.upper()}")

        if ptype == "native_pdf":
            tool_used = "PyMuPDF"
            notes.append("native_text_extracted")
            raw_text = self._extract_native_pdf(path)
        elif ptype == "spreadsheet":
            tool_used = "Pandas/OpenPyXL"
            notes.append("spreadsheet_to_markdown")
            raw_text = self._extract_spreadsheet(path)
        elif ptype in ["scanned_pdf", "image"]:
            vision_model = self._require_vision_model()
            tool_used = f"Local Vision ({vision_model})"
            notes.append("vision_png_standardized")
            notes.append("autorotate_applied") # Traçabilité de l'action géométrique
            notes.append(f"ocr_{vision_model.replace(':', '_')}")
            raw_text = self._extract_vision(path, vision_model)
        elif ptype == "complex_pdf":
            tool_used = "Docling (Fallback PyMuPDF)"
            notes.append("fallback_pymupdf_used")
            raw_text = self._extract_docling(path)
        elif ptype in ("handwritten", "handwriting", "manuscrit"):
            raise ValueError("HANDWRITING_PATH_MISSING")
        else:
            raise ValueError(f"Profil physique inconnu : {ptype}")

        # Construction et retour de l'enveloppe avec son historique complet
        return DocumentResult(
            source_name=path.name,
            profile_type=ptype,
            extraction_tool=tool_used,
            raw_text=raw_text,
            processing_notes=notes,
            metadata={
                "extension": profile.extension,
                "word_count": profile.word_count,
                "has_tables": profile.has_tables,
                "has_images": profile.has_images,
                "is_scanned": profile.is_scanned
            }
        )
        
    def _extract_native_pdf(self, path: Path) -> str:
        print("-> Outil engagé : PyMuPDF (Texte natif)")
        doc = pymupdf.open(path)
        texte = "\n".join([page.get_text() for page in doc])
        doc.close()
        return texte

    def _extract_spreadsheet(self, path: Path) -> str:
        print("-> Outil engagé : Pandas/OpenPyXL (Tableur)")
        if path.suffix.lower() == '.csv':
            df = pd.read_csv(path)
        else:
            df = pd.read_excel(path)
        return df.to_markdown()

    def _require_vision_model(self) -> str:
        if self.vision_model is not None:
            return validate_vision_model(self.vision_model)
        return resolve_model(ModelRole.VISION, ExecutionMode.LOCAL)

    def _extract_vision(self, path: Path, vision_model: str) -> str:
        print(f"-> Outil engagé : Local Vision ({vision_model}) avec AutoRotate")
        doc = pymupdf.open(path)
        
        best_text = ""
        best_len = -1
        best_angle = 0
        
        prompt = (
            "Transcris tout le texte visible. "
            "Si un mot est illisible, écris [?]. "
            "N'invente rien. "
            "Respecte la mise en page."
        )

        # Force Brute : Test des 4 orientations géométriques avec syntaxe robuste
        for angle in [0, 90, 180, 270]:
            matrix = pymupdf.Matrix(1, 1).prerotate(angle)
            
            if path.suffix.lower() == ".pdf":
                pix = doc[0].get_pixmap(matrix=matrix, dpi=200)
            else:
                pix = doc[0].get_pixmap(matrix=matrix)
                
            img_bytes = pix.tobytes("png")
            
            try:
                reponse = ollama.chat(
                    model=vision_model,
                    messages=[{"role": "user", "content": prompt, "images": [img_bytes]}],
                    options={"temperature": 0.0}
                )
                
                texte_courant = reponse["message"]["content"].strip()
                taille_courante = len(texte_courant)
                
                print(f"   [AutoRotate] Test {angle}° -> {taille_courante} caractères")
                
                # On conserve strictement l'orientation produisant le plus de texte
                if taille_courante > best_len:
                    best_len = taille_courante
                    best_text = texte_courant
                    best_angle = angle
            except Exception as e:
                # Si le modèle crash (ex: token repeat limit sur image à l'envers), on ignore et on continue
                print(f"   [AutoRotate] Test {angle}° -> ❌ ÉCHEC du modèle (Ignoré)")

        doc.close()
        print(f"-> Orientation optimale validée : {best_angle}°")
        return best_text

    def _extract_docling(self, path: Path) -> str:
        print("-> Outil engagé : Docling (Tableaux complexes)")
        print("   (Fallback temporaire sur PyMuPDF)")
        return self._extract_native_pdf(path)
